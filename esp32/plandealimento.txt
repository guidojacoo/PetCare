#include <WiFi.h>
#include <WebServer.h>
#include <ESP32Servo.h>
#include <ArduinoJson.h>
#include <time.h>

// =================== WiFi ===================
const char* ssid = "TU_SSID";
const char* password = "TU_PASSWORD";

// =================== Servo ===================
Servo servoMotor;
#define SERVO_PIN 14

// =================== Web Server ===================
WebServer server(8080);

// =================== Variables del plan ===================
int meals = 0;
float secondsOpen = 1;
String mealTimes[4];   // máximo 4 horarios
bool comidasDadas[4];  // para controlar duplicados

// =================== Configuración NTP ===================
const char* ntpServer = "pool.ntp.org";
const long gmtOffset_sec = -10800;  // Argentina GMT-3
const int daylightOffset_sec = 0;

// =================== Utilidad: obtener hora ===================
void printLocalTime() {
  struct tm timeinfo;
  if (!getLocalTime(&timeinfo)) {
    Serial.println("Error obteniendo hora");
    return;
  }
  Serial.println(&timeinfo, "%H:%M:%S");
}

String getCurrentHour() {
  struct tm timeinfo;
  if (!getLocalTime(&timeinfo)) {
    return "";
  }
  char buffer[6];
  strftime(buffer, sizeof(buffer), "%H:%M", &timeinfo);
  return String(buffer);
}

// =================== Endpoint ===================
void handleSetPlan() {
  if (server.hasArg("plain")) {
    StaticJsonDocument<300> doc;
    DeserializationError error = deserializeJson(doc, server.arg("plain"));
    if (!error) {
      meals = doc["meals"];
      secondsOpen = doc["secondsOpen"];

      JsonArray times = doc["times"].as<JsonArray>();
      for (int i = 0; i < meals; i++) {
        mealTimes[i] = times[i].as<String>();
        comidasDadas[i] = false;
      }

      Serial.println("✅ Plan recibido:");
      for (int i = 0; i < meals; i++) {
        Serial.println("Comida " + String(i+1) + " a las " + mealTimes[i]);
      }

      server.send(200, "text/plain", "Plan recibido con horarios");
      return;
    }
  }
  server.send(400, "text/plain", "Error al recibir plan");
}

// =================== Setup ===================
void setup() {
  Serial.begin(115200);

  servoMotor.attach(SERVO_PIN, 500, 2400);
  servoMotor.write(0);

  WiFi.begin(ssid, password);
  Serial.print("Conectando WiFi");
  while (WiFi.status() != WL_CONNECTED) {
    delay(500);
    Serial.print(".");
  }
  Serial.println("\nWiFi conectado: " + WiFi.localIP().toString());

  // Configurar hora NTP
  configTime(gmtOffset_sec, daylightOffset_sec, ntpServer);
  printLocalTime();

  // Endpoint
  server.on("/setPlan", HTTP_POST, handleSetPlan);
  server.begin();
}

// =================== Loop ===================
void loop() {
  server.handleClient();

  if (meals > 0) {
    String horaActual = getCurrentHour();

    for (int i = 0; i < meals; i++) {
      if (!comidasDadas[i] && horaActual == mealTimes[i]) {
        comidasDadas[i] = true;

        Serial.println("🍽 Dando comida #" + String(i+1) + " a las " + horaActual);

        servoMotor.write(90);
        delay((unsigned long)(secondsOpen * 1000));
        servoMotor.write(0);
      }
    }
  }
}
