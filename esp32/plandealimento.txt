#include <WebServer.h>

// ===========================

const char* ssid     = "iPhone de Ignacio";
const char* password = "britos77";

// ===========================
// Servo
// ===========================
Servo servoMotor;
#define SERVO_PIN 14

// ===========================
// Web Server
// ===========================
WebServer server(8080);

// ===========================
// Plan semanal
// ===========================
// Máximo de horarios por día (ajustá si querés)
#define MAX_MEALS_PER_DAY 12

// Guardamos horas como "HH:MM"
String dayTimes[7][MAX_MEALS_PER_DAY];
int    dayCount[7] = {0,0,0,0,0,0,0};
bool   fed[7][MAX_MEALS_PER_DAY];  // si ya se dio esa comida
float  secondsOpen = 1.0;

// Para detectar cambio de día
int lastWday = -1;

// Último evento para el monitor
int    lastFeedWday  = -1;     // 0=Domingo..6=Sábado
int    lastFeedIndex = -1;     // índice de la comida del día
String lastFeedTime  = "";     // HH:MM del disparo

// ===========================
// NTP (hora/día)
// ===========================
const char* ntpServer = "pool.ntp.org";
const long gmtOffset_sec = -10800;  // Argentina GMT-3
const int daylightOffset_sec = 0;

String getCurrentHHMM() {
  struct tm timeinfo;
  if (!getLocalTime(&timeinfo)) return "";
  char buf[6];
  strftime(buf, sizeof(buf), "%H:%M", &timeinfo);
  return String(buf);
}

// tm_wday: 0=Domingo, 1=Lunes, ... 6=Sábado
int getWday() {
  struct tm timeinfo;
  if (!getLocalTime(&timeinfo)) return -1;
  return timeinfo.tm_wday;
}

// ===========================
// CORS
// ===========================
void enableCORS() {
  server.sendHeader("Access-Control-Allow-Origin", "*");
  server.sendHeader("Access-Control-Allow-Methods", "GET, POST, OPTIONS");
  server.sendHeader("Access-Control-Allow-Headers", "Content-Type");
}
void handleOptions() {
  enableCORS();
  server.send(204);
}

// ===========================
// Utilidades
// ===========================
void resetFedForDay(int wday) {
  if (wday < 0 || wday > 6) return;
  for (int i = 0; i < dayCount[wday]; i++) {
    fed[wday][i] = false;
  }
}

void resetAllFed() {
  for (int d = 0; d < 7; d++) resetFedForDay(d);
}

bool isValidHHMM(const String& s) {
  if (s.length() != 5) return false;
  if (s.charAt(2) != ':') return false;
  int hh = s.substring(0,2).toInt();
  int mm = s.substring(3,5).toInt();
  return (hh >= 0 && hh <= 23 && mm >= 0 && mm <= 59);
}

String getDayName(int wday) {
  // 0=Domingo..6=Sábado
  const char* names[7] = {"Domingo","Lunes","Martes","Miércoles","Jueves","Viernes","Sábado"};
  if (wday < 0 || wday > 6) return "Desconocido";
  return String(names[wday]);
}

// ===========================
// Endpoint: /status (GET)
// ===========================
void handleStatus() {
  enableCORS();

  struct tm timeinfo;
  if (!getLocalTime(&timeinfo)) {
    server.send(503, "application/json", "{\"ok\":false,\"error\":\"no_time\"}");
    return;
  }

  int wday = timeinfo.tm_wday;  // 0..6
  char nowbuf[6];
  strftime(nowbuf, sizeof(nowbuf), "%H:%M", &timeinfo);
  String nowHHMM = String(nowbuf);

  String json = "{";
  json += "\"ok\":true,";
  json += "\"now\":{\"wday\":" + String(wday) + ",\"dayName\":\"" + getDayName(wday) + "\",\"hhmm\":\"" + nowHHMM + "\"},";
  json += "\"secondsOpen\":" + String(secondsOpen, 2) + ",";

  // Último evento
  json += "\"lastFeed\":{";
  json += "\"wday\":" + String(lastFeedWday) + ",";
  json += "\"dayName\":\"" + ((lastFeedWday>=0)?getDayName(lastFeedWday):String("")) + "\",";
  json += "\"index\":" + String(lastFeedIndex) + ",";
  json += "\"hhmm\":\"" + lastFeedTime + "\"},";

  // Plan de hoy
  json += "\"today\":{";
  json += "\"wday\":" + String(wday) + ",";
  json += "\"dayName\":\"" + getDayName(wday) + "\",";
  json += "\"total\":" + String(dayCount[wday]) + ",";
  json += "\"plan\":[";
  for (int i = 0; i < dayCount[wday]; i++) {
    if (i>0) json += ",";
    json += "{";
    json += "\"index\":" + String(i) + ",";
    json += "\"time\":\"" + dayTimes[wday][i] + "\",";
    json += "\"fed\":" + String(fed[wday][i] ? "true" : "false");
    json += "}";
  }
  json += "]}";
  json += "}";

  server.send(200, "application/json", json);
}

// ===========================
// Endpoint: /setPlan  (POST)
// body: { "secondsOpen": <float>, "weekplan": [ [..lunes..], [..martes..], ..., [..domingo..] ] }
// NOTA: El orden que llega es Lunes..Domingo
// Internamente convertimos a 0=Domingo..6=Sábado para comparar con tm_wday.
// ===========================
void handleSetPlan() {
  if (!server.hasArg("plain")) {
    enableCORS();
    server.send(400, "text/plain", "Falta cuerpo JSON");
    return;
  }

  StaticJsonDocument<1500> doc; // tamaño suficiente para listas
  DeserializationError err = deserializeJson(doc, server.arg("plain"));
  if (err) {
    enableCORS();
    server.send(400, "text/plain", "JSON invalido");
    return;
  }

  if (!doc.containsKey("secondsOpen") || !doc.containsKey("weekplan")) {
    enableCORS();
    server.send(400, "text/plain", "Campos requeridos: secondsOpen, weekplan");
    return;
  }

  secondsOpen = doc["secondsOpen"].as<float>();

  // weekplan llega como 7 arrays: 0=Lunes .. 6=Domingo (ISO)
  JsonArray week = doc["weekplan"].as<JsonArray>();
  if (week.size() != 7) {
    enableCORS();
    server.send(400, "text/plain", "weekplan debe tener 7 elementos");
    return;
  }

  // Limpio
  for (int d = 0; d < 7; d++) {
    dayCount[d] = 0;
  }

  // Mapeo: ISO (Lun..Dom) -> tm_wday (Dom..Sab)
  auto isoToWday = [](int iso)->int {
    if (iso == 6) return 0;     // Domingo
    return iso + 1;             // Lunes..Sábado -> 1..6
  };

  for (int iso = 0; iso < 7; iso++) {
    int wday = isoToWday(iso);
    JsonArray times = week[iso].as<JsonArray>();
    int idx = 0;
    for (JsonVariant v : times) {
      if (idx >= MAX_MEALS_PER_DAY) break;
      String t = v.as<String>();
      t.trim();
      if (isValidHHMM(t)) {
        dayTimes[wday][idx] = t;
        idx++;
      }
    }
    dayCount[wday] = idx;
  }

  // Al cargar nuevo plan: resetear banderas de comidas del día actual
  resetAllFed();

  // Log
  Serial.println("=== Plan semanal recibido ===");
  Serial.printf("secondsOpen: %.2f s\n", secondsOpen);
  for (int d = 0; d < 7; d++) {
    Serial.printf("Dia %d (tm_wday): ", d);
    for (int i = 0; i < dayCount[d]; i++) {
      Serial.print(dayTimes[d][i]);
      if (i < dayCount[d]-1) Serial.print(", ");
    }
    Serial.println();
  }

  enableCORS();
  server.send(200, "text/plain", "Plan semanal cargado");
}

// (Opcional) borrar plan rápido
void handleClearPlan() {
  for (int d=0; d<7; d++) {
    dayCount[d] = 0;
    for (int i=0; i<MAX_MEALS_PER_DAY; i++) fed[d][i] = false;
  }
  lastFeedWday = -1;
  lastFeedIndex = -1;
  lastFeedTime = "";
  enableCORS();
  server.send(200, "text/plain", "Plan borrado");
}

void handleServoManual() {
  enableCORS();
  servoMotor.write(90);      // abrir
  delay(2000);               // esperar 2 segundos
  servoMotor.write(0);       // volver a cerrado
  server.send(200, "text/plain", "Servo movido manualmente");
}

// ===========================
// Setup
// ===========================
void setup() {
  Serial.begin(115200);

  // Servo
  servoMotor.attach(SERVO_PIN, 500, 2400);
  servoMotor.write(0);

  // WiFi
  WiFi.begin(ssid, password);
  Serial.print("Conectando WiFi");
  while (WiFi.status() != WL_CONNECTED) {
    delay(500);
    Serial.print(".");
  }
  Serial.println("\nWiFi OK: " + WiFi.localIP().toString());

  // Hora NTP
  configTime(gmtOffset_sec, daylightOffset_sec, ntpServer);
  struct tm timeinfo;
  if (getLocalTime(&timeinfo)) {
    lastWday = timeinfo.tm_wday;
    Serial.printf("Hora inicial: %02d:%02d  wday=%d\n", timeinfo.tm_hour, timeinfo.tm_min, lastWday);
  } else {
    Serial.println("No se pudo obtener hora inicial (NTP)");
  }

  // Endpoints
  server.on("/setPlan",   HTTP_POST,    handleSetPlan);
  server.on("/setPlan",   HTTP_OPTIONS, handleOptions);

  server.on("/status",    HTTP_GET,     handleStatus);
  server.on("/status",    HTTP_OPTIONS, handleOptions);

  server.on("/clearPlan", HTTP_POST,    handleClearPlan);
  server.on("/clearPlan", HTTP_OPTIONS, handleOptions);

  server.on("/servo",     HTTP_GET,     handleServoManual);
  server.on("/servo",     HTTP_OPTIONS, handleOptions);
  
  server.begin();

  // Estado inicial: sin comidas dadas
  resetAllFed();
}

// ===========================
// Loop
// ===========================
unsigned long lastCheck = 0;
const unsigned long CHECK_EVERY_MS = 1000; // chequeo cada 1s

void loop() {
  server.handleClient();

  // Chequeo periódico
  unsigned long now = millis();
  if (now - lastCheck < CHECK_EVERY_MS) return;
  lastCheck = now;

  // Obtener hora/día actual
  struct tm timeinfo;
  if (!getLocalTime(&timeinfo)) {
    // Si no hay hora, no alimentamos
    return;
  }

  int wday = timeinfo.tm_wday; // 0=Domingo..6=Sábado
  // Cambio de día → limpiar "fed" para el nuevo día
  if (wday != lastWday) {
    resetFedForDay(wday);
    lastWday = wday;
    Serial.printf("Cambio de dia: wday=%d, reset flags\n", wday);
  }

  // Formato HH:MM actual
  char hmbuf[6];
  strftime(hmbuf, sizeof(hmbuf), "%H:%M", &timeinfo);
  String nowHHMM = String(hmbuf);

  // ¿Hay plan para este día?
  int count = dayCount[wday];
  for (int i = 0; i < count; i++) {
    if (!fed[wday][i] && nowHHMM == dayTimes[wday][i]) {
      // Disparamos servo una sola vez por horario
      fed[wday][i] = true;
      Serial.printf("Alimentando (wday=%d) a las %s\n", wday, dayTimes[wday][i].c_str());

      // Registrar último evento para /status
      lastFeedWday  = wday;
      lastFeedIndex = i;
      lastFeedTime  = nowHHMM;

      servoMotor.write(90); // abrir
      delay((unsigned long)(secondsOpen * 1000));
      servoMotor.write(0);  // cerrar
    }
  }
}
