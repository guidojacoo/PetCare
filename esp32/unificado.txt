#define PIR_PIN 12                 // OUT del PIR ahora en GPIO12

// ===========================
// Bomba (relé/MOSFET de control)
// ===========================
#define PUMP_PIN 25
// Relé ACTIVO EN LOW:
const uint8_t PUMP_ON_LEVEL  = LOW;   // LOW = relé activado
const uint8_t PUMP_OFF_LEVEL = HIGH;  // HIGH = relé desactivado
bool pumpOn = false;                  // estado lógico de la bomba

// Espera entre servo y bomba (para no superponer consumo)
const unsigned long PUMP_GAP_MS = 2000; // 2 s

// Helpers de bomba
inline void pump_on()  { digitalWrite(PUMP_PIN, PUMP_ON_LEVEL);  pumpOn = true;  }
inline void pump_off() { digitalWrite(PUMP_PIN, PUMP_OFF_LEVEL); pumpOn = false; }

// ===========================
// Web Server
// ===========================
WebServer server(8080);

// ===========================
// Plan (por día o por semana)
// ===========================
#define MAX_MEALS_PER_DAY 12

String dayTimes[7][MAX_MEALS_PER_DAY];
int    dayCount[7] = {0,0,0,0,0,0,0};
bool   fed[7][MAX_MEALS_PER_DAY];
bool   eaten[7][MAX_MEALS_PER_DAY];
bool   water[7][MAX_MEALS_PER_DAY];    // agua servida
float  secondsOpen = 1.0;

int lastWday = -1;

// Último evento
int    lastFeedWday  = -1;
int    lastFeedIndex = -1;
String lastFeedTime  = "";

// Estado de espera de "comido"
bool           waitingEat       = false;
int            pendingEatWday   = -1;
int            pendingEatIndex  = -1;
unsigned long  eatWindowStartMs = 0;

// ===========================
// NTP
// ===========================
const char* ntpServer = "pool.ntp.org";
const long gmtOffset_sec = -10800;
const int  daylightOffset_sec = 0;

// ===========================
// CORS
// ===========================
void enableCORS() {
  server.sendHeader("Access-Control-Allow-Origin", "*");
  server.sendHeader("Access-Control-Allow-Methods", "GET, POST, OPTIONS");
  server.sendHeader("Access-Control-Allow-Headers", "Content-Type");
}
void handleOptions() { enableCORS(); server.send(204); }

// ===========================
// NVS (Preferences) para persistir plan y ticks
// ===========================
Preferences prefs;

// Guarda el JSON crudo recibido en /setPlan
void savePlanJSONToNVS(const String& json) {
  if (prefs.begin("petcare", false)) {
    prefs.putString("weekplan_json", json);
    prefs.end();
  }
}

// Lee el último JSON de plan guardado (vacío si no hay)
String loadPlanJSONFromNVS() {
  String out;
  if (prefs.begin("petcare", true)) {
    out = prefs.getString("weekplan_json", "");
    prefs.end();
  }
  return out;
}

// ===== Persistencia de flags (fed/eaten/water) por día =====
void saveDayFlags(int wday) {
  if (wday < 0 || wday > 6) return;
  if (prefs.begin("petcare", false)) {
    String kf = "fed_"    + String(wday);
    String ke = "eaten_"  + String(wday);
    String kw = "water_"  + String(wday);
    prefs.putBytes(kf.c_str(),   fed[wday],   MAX_MEALS_PER_DAY);
    prefs.putBytes(ke.c_str(),   eaten[wday], MAX_MEALS_PER_DAY);
    prefs.putBytes(kw.c_str(),   water[wday], MAX_MEALS_PER_DAY);
    prefs.end();
  }
}

void loadDayFlags(int wday) {
  if (wday < 0 || wday > 6) return;
  if (prefs.begin("petcare", true)) {
    String kf = "fed_"    + String(wday);
    String ke = "eaten_"  + String(wday);
    String kw = "water_"  + String(wday);
    // Si no existían, getBytes devuelve 0 y mantenemos los false por defecto
    prefs.getBytes(kf.c_str(),   fed[wday],   MAX_MEALS_PER_DAY);
    prefs.getBytes(ke.c_str(),   eaten[wday], MAX_MEALS_PER_DAY);
    prefs.getBytes(kw.c_str(),   water[wday], MAX_MEALS_PER_DAY);
    prefs.end();
  }
}

void saveAllFlags() {
  for (int d=0; d<7; d++) saveDayFlags(d);
}
void loadAllFlags() {
  for (int d=0; d<7; d++) loadDayFlags(d);
}

// ===========================
// Utils
// ===========================
bool isValidHHMM(const String& s) {
  if (s.length() != 5 || s.charAt(2) != ':') return false;
  int hh = s.substring(0,2).toInt();
  int mm = s.substring(3,5).toInt();
  return (hh>=0 && hh<=23 && mm>=0 && mm<=59);
}
void resetFedForDay(int wday) {
  if (wday < 0 || wday > 6) return;
  for (int i = 0; i < dayCount[wday]; i++) {
    fed[wday][i]   = false;
    eaten[wday][i] = false;
    water[wday][i] = false;    // reset agua
  }
}
void resetAllFed() { for (int d=0; d<7; d++) resetFedForDay(d); }
String getDayName(int wday) {
  const char* names[7]={"Domingo","Lunes","Martes","Miércoles","Jueves","Viernes","Sábado"};
  if (wday<0||wday>6) return "Desconocido"; return String(names[wday]);
}

// ===========================
// /status
// ===========================
void handleStatus() {
  enableCORS();

  struct tm timeinfo;
  if (!getLocalTime(&timeinfo)) {
    server.send(503, "application/json", "{\"ok\":false,\"error\":\"no_time\"}");
    return;
  }
  int wday = timeinfo.tm_wday;
  char nowbuf[6]; strftime(nowbuf, sizeof(nowbuf), "%H:%M", &timeinfo);

  String json = "{";
  json += "\"ok\":true,";
  json += "\"now\":{\"wday\":" + String(wday) + ",\"dayName\":\"" + getDayName(wday) + "\",\"hhmm\":\"" + String(nowbuf) + "\"},";
  json += "\"secondsOpen\":" + String(secondsOpen,2) + ",";
  json += "\"eat_waiting\":" + String(waitingEat ? "true":"false") + ",";
  json += "\"pumpOn\":" + String(pumpOn ? "true":"false") + ",";

  json += "\"lastFeed\":{";
  json += "\"wday\":" + String(lastFeedWday) + ",";
  json += "\"dayName\":\"" + ((lastFeedWday>=0)?getDayName(lastFeedWday):String("")) + "\",";
  json += "\"index\":" + String(lastFeedIndex) + ",";
  json += "\"hhmm\":\"" + lastFeedTime + "\"},";

  json += "\"today\":{";
  json += "\"wday\":" + String(wday) + ",";
  json += "\"dayName\":\"" + getDayName(wday) + "\",";
  json += "\"total\":" + String(dayCount[wday]) + ",";
  json += "\"plan\":[";
  for (int i = 0; i < dayCount[wday]; i++) {
    if (i) json += ",";
    json += "{";
    json += "\"index\":" + String(i) + ",";
    json += "\"time\":\"" + dayTimes[wday][i] + "\",";
    json += "\"fed\":" + String(fed[wday][i] ? "true":"false") + ",";
    json += "\"eaten\":" + String(eaten[wday][i] ? "true":"false") + ",";
    json += "\"water\":" + String(water[wday][i] ? "true":"false");
    json += "}";
  }
  json += "]}";
  json += "}";
  server.send(200, "application/json", json);
}

// ===========================
// /setPlan  (POST)
// ===========================
void handleSetPlan() {
  if (!server.hasArg("plain")) { enableCORS(); server.send(400,"text/plain","Falta cuerpo JSON"); return; }

  StaticJsonDocument<2000> doc;
  DeserializationError err = deserializeJson(doc, server.arg("plain"));
  if (err) { enableCORS(); server.send(400,"text/plain","JSON invalido"); return; }

  if (!doc.containsKey("secondsOpen")) { enableCORS(); server.send(400,"text/plain","Falta secondsOpen"); return; }
  secondsOpen = doc["secondsOpen"].as<float>();

  String mode = doc.containsKey("mode") ? String(doc["mode"].as<const char*>()) : "per_day";

  // limpiar plan actual
  for (int d=0; d<7; d++){
    dayCount[d]=0;
    for(int i=0;i<MAX_MEALS_PER_DAY;i++){
      fed[d][i]=false; eaten[d][i]=false; water[d][i]=false;
    }
  }

  auto isoToWday = [](int iso)->int { return (iso==6)?0:(iso+1); }; // ISO Lun..Dom -> tm_wday Dom..Sáb

  bool ok=false;
  if (mode=="repeat_week") {
    if (!doc.containsKey("dayplan")) { enableCORS(); server.send(400,"text/plain","Falta dayplan en repeat_week"); return; }
    JsonArray arr = doc["dayplan"].as<JsonArray>();
    String tmp[MAX_MEALS_PER_DAY]; int cnt=0;
    for (JsonVariant v: arr){ if (cnt>=MAX_MEALS_PER_DAY) break; String t=v.as<String>(); t.trim(); if (isValidHHMM(t)) tmp[cnt++]=t; }
    for (int d=0; d<7; d++){ for (int i=0;i<cnt;i++){ dayTimes[d][i]=tmp[i]; } dayCount[d]=cnt; }
    ok=true;
  } else {
    if (!doc.containsKey("weekplan")) { enableCORS(); server.send(400,"text/plain","Falta weekplan en per_day"); return; }
    JsonArray week = doc["weekplan"].as<JsonArray>();
    if (week.size()!=7){ enableCORS(); server.send(400,"text/plain","weekplan debe tener 7 elementos"); return; }
    for (int iso=0; iso<7; iso++){
      int wday=isoToWday(iso);
      JsonArray times = week[iso].as<JsonArray>();
      int idx=0;
      for (JsonVariant v: times){
        if (idx>=MAX_MEALS_PER_DAY) break;
        String t=v.as<String>(); t.trim();
        if (isValidHHMM(t)) { dayTimes[wday][idx]=t; idx++; }
      }
      dayCount[wday]=idx;
    }
    ok=true;
  }

  resetAllFed();
  waitingEat=false; pendingEatWday=-1; pendingEatIndex=-1;

  // Persistir plan
  savePlanJSONToNVS(server.arg("plain"));

  // Persistir reset de flags
  saveAllFlags();

  Serial.println("=== Plan cargado ===");
  Serial.printf("mode: %s  secondsOpen: %.2f s\n", mode.c_str(), secondsOpen);
  for (int d=0; d<7; d++){ Serial.printf("Dia %d: ", d); for(int i=0;i<dayCount[d];i++){ Serial.print(dayTimes[d][i]); if(i<dayCount[d]-1) Serial.print(", "); } Serial.println(); }

  enableCORS();
  if (ok) server.send(200,"text/plain","Plan cargado");
  else    server.send(400,"text/plain","Error al cargar plan");
}

// ===========================
// /clearPlan (POST)
// ===========================
void handleClearPlan() {
  for (int d=0; d<7; d++){
    dayCount[d]=0;
    for(int i=0;i<MAX_MEALS_PER_DAY;i++){
      fed[d][i]=false; eaten[d][i]=false; water[d][i]=false;
    }
    saveDayFlags(d); // persistir reset de cada día
  }
  lastFeedWday=-1; lastFeedIndex=-1; lastFeedTime="";
  waitingEat=false; pendingEatWday=-1; pendingEatIndex=-1;
  enableCORS(); server.send(200,"text/plain","Plan borrado");
}

// ===========================
// /servo (GET) - manual
//   servo -> gap 2s -> bomba (secondsOpen)
// ===========================
void handleServoManual() {
  enableCORS();

  servoMotor.write(90);
  delay(2000);
  servoMotor.write(0);

  server.send(200, "text/plain", "Servo ");
}

// ===========================
// /getPlan (GET)
// ===========================
void handleGetPlan() {
  enableCORS();
  String json = loadPlanJSONFromNVS();
  if (json.length() == 0) {
    // Construimos uno por defecto desde el estado actual (per_day)
    StaticJsonDocument<2500> doc;
    doc["ok"] = true;
    doc["mode"] = "per_day";
    doc["secondsOpen"] = secondsOpen;

    JsonArray week = doc.createNestedArray("weekplan");
    // Orden ISO (Lun..Dom) => indices 0..6
    for (int iso=0; iso<7; ++iso) {
      int wday = (iso == 6) ? 0 : iso + 1; // inversa de isoToWday
      JsonArray arr = week.createNestedArray();
      for (int i=0; i<dayCount[wday]; ++i) {
        arr.add(dayTimes[wday][i]);
      }
    }

    String out;
    serializeJson(doc, out);
    server.send(200, "application/json", out);
    return;
  }

  server.send(200, "application/json", json);
}

// ===========================
// Setup
// ===========================
void setup() {
  Serial.begin(115200);

  servoMotor.attach(SERVO_PIN, 500, 2400);
  servoMotor.write(0);

  pinMode(PIR_PIN, INPUT);     // HC-SR501 OUT

  // Mantener la bomba APAGADA al arrancar (relé activo en LOW)
  pinMode(PUMP_PIN, INPUT_PULLUP);      // ayuda a que arranque en HIGH
  delay(5);
  pinMode(PUMP_PIN, OUTPUT);
  pump_off(); // = digitalWrite(PUMP_PIN, PUMP_OFF_LEVEL)

  WiFi.begin(ssid, password);
  Serial.print("Conectando WiFi");
  while (WiFi.status()!=WL_CONNECTED){ delay(500); Serial.print("."); }
  Serial.println("\nWiFi OK: " + WiFi.localIP().toString());

  configTime(gmtOffset_sec, daylightOffset_sec, ntpServer);
  struct tm timeinfo;
  if (getLocalTime(&timeinfo)) { lastWday = timeinfo.tm_wday; Serial.printf("Hora inicial: %02d:%02d  wday=%d\n", timeinfo.tm_hour, timeinfo.tm_min, lastWday); }
  else                          { Serial.println("No se pudo obtener hora inicial (NTP)"); }

  // Restaurar plan desde NVS
  {
    String saved = loadPlanJSONFromNVS();
    if (saved.length() > 0) {
      StaticJsonDocument<2500> doc;
      DeserializationError err = deserializeJson(doc, saved);
      if (!err) {
        if (doc.containsKey("secondsOpen")) {
          secondsOpen = doc["secondsOpen"].as<float>();
        }
        String mode = doc.containsKey("mode") ? String(doc["mode"].as<const char*>()) : "per_day";

        // Limpia plan actual
        for (int d=0; d<7; d++){
          dayCount[d]=0;
          for(int i=0;i<MAX_MEALS_PER_DAY;i++){
            fed[d][i]=false; eaten[d][i]=false; water[d][i]=false;
          }
        }

        auto isoToWday = [](int iso)->int { return (iso==6)?0:(iso+1); };

        if (mode=="repeat_week" && doc.containsKey("dayplan")) {
          JsonArray arr = doc["dayplan"].as<JsonArray>();
          String tmp[MAX_MEALS_PER_DAY]; int cnt=0;
          for (JsonVariant v: arr){ if (cnt>=MAX_MEALS_PER_DAY) break; String t=v.as<String>(); t.trim(); if (isValidHHMM(t)) tmp[cnt++]=t; }
          for (int d=0; d<7; d++){ for (int i=0;i<cnt;i++){ dayTimes[d][i]=tmp[i]; } dayCount[d]=cnt; }
        } else if (mode=="per_day" && doc.containsKey("weekplan")) {
          JsonArray week = doc["weekplan"].as<JsonArray>();
          if (week.size()==7){
            for (int iso=0; iso<7; iso++){
              int wday=isoToWday(iso);
              JsonArray times = week[iso].as<JsonArray>();
              int idx=0;
              for (JsonVariant v: times){
                if (idx>=MAX_MEALS_PER_DAY) break;
                String t=v.as<String>(); t.trim();
                if (isValidHHMM(t)) { dayTimes[wday][idx]=t; idx++; }
              }
              dayCount[wday]=idx;
            }
          }
        }
        resetAllFed();
        waitingEat=false; pendingEatWday=-1; pendingEatIndex=-1;

        Serial.println("Plan restaurado desde NVS.");
      } else {
        Serial.println("No se pudo parsear plan NVS.");
      }
    } else {
      Serial.println("No hay plan NVS guardado.");
    }
  }

  // Cargar flags previos (si existían)
  loadAllFlags();

  // Rutas HTTP
  server.on("/setPlan",   HTTP_POST,    handleSetPlan);
  server.on("/setPlan",   HTTP_OPTIONS, handleOptions);
  server.on("/status",    HTTP_GET,     handleStatus);
  server.on("/status",    HTTP_OPTIONS, handleOptions);
  server.on("/clearPlan", HTTP_POST,    handleClearPlan);
  server.on("/clearPlan", HTTP_OPTIONS, handleOptions);
  server.on("/servo",     HTTP_GET,     handleServoManual);
  server.on("/servo",     HTTP_OPTIONS, handleOptions);

  // Endpoint para obtener el último plan guardado
  server.on("/getPlan",   HTTP_GET,     handleGetPlan);
  server.on("/getPlan",   HTTP_OPTIONS, handleOptions);

  server.begin();
  resetAllFed();
}

// ===========================
// Loop
// ===========================
unsigned long lastCheck = 0;
const unsigned long CHECK_EVERY_MS = 1000;

void loop() {
  server.handleClient();

  // Chequeo plan cada 1s
  unsigned long nowMs = millis();
  if (nowMs - lastCheck >= CHECK_EVERY_MS) {
    lastCheck = nowMs;

    struct tm timeinfo;
    if (getLocalTime(&timeinfo)) {
      int wday = timeinfo.tm_wday;

      if (wday != lastWday) {
        resetFedForDay(wday);
        lastWday = wday;
        saveDayFlags(wday); // persistir reset del nuevo día
        Serial.printf("Cambio de dia: wday=%d, reset flags\n", wday);
      }

      char hmbuf[6]; strftime(hmbuf, sizeof(hmbuf), "%H:%M", &timeinfo);
      String nowHHMM = String(hmbuf);

      int count = dayCount[wday];
      for (int i=0; i<count; i++) {
        if (!fed[wday][i] && nowHHMM == dayTimes[wday][i]) {
          fed[wday][i] = true;

          lastFeedWday  = wday;
          lastFeedIndex = i;
          lastFeedTime  = nowHHMM;

          Serial.printf("Alimentando (wday=%d) a las %s\n", wday, dayTimes[wday][i].c_str());

          // === Secuencia SECUENCIAL para bajar pico de consumo ===
          // 1) Servo por secondsOpen
          servoMotor.write(90);
          delay((unsigned long)(secondsOpen * 1000));
          servoMotor.write(0);

          // 2) Gap
          delay(PUMP_GAP_MS);

          // 3) Bomba por secondsOpen (activo en LOW)
          pump_on();
          delay((unsigned long)(secondsOpen * 1000));
          pump_off();

          // Marcar agua servida
          water[wday][i] = true;

          // Guardar cambios del día (fed/agua)
          saveDayFlags(wday);

          // Abrir ventana PIR para "comido"
          waitingEat       = true;
          pendingEatWday   = wday;
          pendingEatIndex  = i;
          eatWindowStartMs = millis();
        }
      }
    }
  }

  // Lógica PIR: si hay ventana, marcar "eaten" al primer HIGH
  if (waitingEat) {
    if (millis() - eatWindowStartMs >= EAT_WINDOW_MS) {
      waitingEat = false; // se cerró la ventana sin detección
    } else {
      if (digitalRead(PIR_PIN) == HIGH) {
        if (pendingEatWday>=0 && pendingEatIndex>=0 &&
            pendingEatWday<=6 && pendingEatIndex<dayCount[pendingEatWday]) {
          eaten[pendingEatWday][pendingEatIndex] = true;
          Serial.printf("Comido detectado (wday=%d, idx=%d)\n", pendingEatWday, pendingEatIndex);

          // Guardar cambios del día (eaten)
          saveDayFlags(pendingEatWday);
        }
        waitingEat = false; // cerrar ventana al primer movimiento
      }
    }
  }
}
