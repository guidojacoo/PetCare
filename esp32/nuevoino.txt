bool isValidHHMM(const String& s) {
  if (s.length() != 5 || s.charAt(2) != ':') return false;
  int hh = s.substring(0,2).toInt();
  int mm = s.substring(3,5).toInt();
  return (hh>=0 && hh<=23 && mm>=0 && mm<=59);

int hhmmToMin(const String& s){
  if (s.length()!=5) return -1;
  int hh = s.substring(0,2).toInt();
  int mm = s.substring(3,5).toInt();
  return (hh>=0 && hh<24 && mm>=0 && mm<60) ? (hh*60+mm) : -1;
}
bool isNowWithinBlackout(){
  struct tm ti;
  if (!getLocalTime(&ti)) return false;
  int wday = ti.tm_wday;
  int nowMin = ti.tm_hour*60 + ti.tm_min;
  unsigned long preS  = MIC_BLACKOUT_BEFORE_MS/1000UL;
  unsigned long postS = MIC_BLACKOUT_AFTER_MS/1000UL;
  for (int i=0;i<dayCount[wday];++i){
    int m = hhmmToMin(dayTimes[wday][i]);
    if (m<0) continue;
    int from = m - (int)(preS/60UL);
    int to   = m + (int)(postS/60UL);
    if (nowMin >= from && nowMin <= to) return true;
  }
  return false;
}

void micTask(void*){
  for(;;){
    if (!micEnabled){ vTaskDelay(pdMS_TO_TICKS(50)); continue; }
    if (micStopAtMs!=0 && millis() > micStopAtMs){
      micEnabled = false;
      micStopAtMs = 0;
      vTaskDelay(pdMS_TO_TICKS(50));
      continue;
    }
    micSuspendedByPlan = isNowWithinBlackout();
    if (micSuspendedByPlan){ vTaskDelay(pdMS_TO_TICKS(500)); continue; }

    float mean=0.0f, m2=0.0f, peak=0.0f;
    for (int i=0;i<MIC_WINDOW_SAMPLES;i++){
      int raw = analogRead(MIC_PIN);
      float x = (float)raw;
      float d = x - mean; mean += d/(i+1);   m2 += d*(x-mean);
      float a = fabsf(x - mean); if(a>peak) peak=a;
      delayMicroseconds(MIC_SAMPLE_US);
    }
    float variance = (MIC_WINDOW_SAMPLES>1) ? (m2/(MIC_WINDOW_SAMPLES-1)) : 0.0f;
    float rms = sqrtf(variance);

    float lvl01 = rms  / 1200.0f; if (lvl01>1) lvl01=1;
    float pk01  = peak / 2048.0f; if (pk01 >1) pk01 =1;

    micLevel01 = lvl01;
    micPeak01  = pk01;

    int lvl100 = (int)lroundf(lvl01*100.0f);
    int pk100  = (int)lroundf(pk01 *100.0f);
    bool high = (lvl100 >= micThreshold) || (pk100 >= micThreshold);

    if (high && !micInAlert){
      micAlertsToday++;
      micInAlert=true;
      struct tm ti;
      if (getLocalTime(&ti)) {
        micAlerts[ti.tm_wday] = micAlertsToday;
        unsigned long now = millis();
        if (now - micPersistMs > 60000UL) {
          saveMicAlerts(ti.tm_wday);
          micPersistMs = now;
        }
      }
    }
    if (!high && micInAlert){ micInAlert=false; }

    vTaskDelay(1);
  }
}

uint32_t djb2hash(const String& s){
  uint32_t h = 5381;
  for (size_t i=0;i<s.length();++i) h = ((h<<5)+h) + (uint8_t)s[i];
  return h ? h : 1;
}
String calibKeyForUser(const String& userId){ char buf[16]; snprintf(buf, sizeof(buf), "cal_u_%08lx", (unsigned long)djb2hash(userId)); return String(buf); }
String waterKeyForUser(const String& userId){ char buf[16]; snprintf(buf, sizeof(buf), "wat_u_%08lx", (unsigned long)djb2hash(userId)); return String(buf); }
String flowKeyForUser (const String& userId){ char buf[16]; snprintf(buf, sizeof(buf), "flo_u_%08lx", (unsigned long)djb2hash(userId)); return String(buf); }

void handleSetPlan() {
  if (!server.hasArg("plain")) { enableCORS(); server.send(400,"text/plain","Falta cuerpo JSON"); return; }

  StaticJsonDocument<2000> doc;
  DeserializationError err = deserializeJson(doc, server.arg("plain"));
  if (err) { enableCORS(); server.send(400,"text/plain","JSON invalido"); return; }

  if (!doc.containsKey("secondsOpen")) { enableCORS(); server.send(400,"text/plain","Falta secondsOpen"); return; }
  secondsOpen = doc["secondsOpen"].as<float>();

  // NUEVO: userId + agua/flujo (persistir por usuario)
  String planUser = doc["userId"] | "";
  if (planUser.length()>0) {
    currentPlanUserId = planUser;
    if (prefs.begin("petcare", false)) { prefs.putString("plan_user", currentPlanUserId); prefs.end(); }
  }
  if (doc.containsKey("water_ml_per_meal")) {
    float ml = doc["water_ml_per_meal"].as<float>();
    if (planUser.length()>0 && isfinite(ml) && ml>=0) saveUserWaterML(planUser, ml);
  }
  if (doc.containsKey("pump_ml_per_sec")) {
    float r = doc["pump_ml_per_sec"].as<float>();
    if (planUser.length()>0 && isfinite(r) && r>0) saveUserPumpFlow(planUser, r);
  }

  String mode = doc.containsKey("mode") ? String(doc["mode"].as<const char*>()) : "per_day";

  for (int d=0; d<7; d++){
    dayCount[d]=0;
    for(int i=0;i<MAX_MEALS_PER_DAY;i++){
      fed[d][i]=false; eaten[d][i]=false; water[d][i]=false;
    }
  }

  auto isoToWday = [](int iso)->int { return (iso==6)?0:(iso+1); };

  bool ok=false;
  if (mode=="repeat_week") {
    if (!doc.containsKey("dayplan")) { enableCORS(); server.send(400,"text/plain","Falta dayplan en repeat_week"); return; }
    JsonArray arr = doc["dayplan"].as<JsonArray>();
    String tmp[MAX_MEALS_PER_DAY]; int cnt=0;
    for (JsonVariant v: arr){ if (cnt>=MAX_MEALS_PER_DAY) break; String t=v.as<String>(); t.trim(); if (isValidHHMM(t)) tmp[cnt++]=t; }
    for (int d=0; d<7; d++){ for (int i=0;i<cnt;i++){ dayTimes[d][i]=tmp[i]; } dayCount[d]=cnt; }
    ok=true;
  } else {
    if (!doc.containsKey("weekplan")) { enableCORS(); server.send(400,"text/plain","Falta weekplan en per_day"); return; }
    JsonArray week = doc["weekplan"].as<JsonArray>();
    if (week.size()!=7){ enableCORS(); server.send(400,"text/plain","weekplan debe tener 7 elementos"); return; }
    for (int iso=0; iso<7; iso++){
      int wday=isoToWday(iso);
      JsonArray times = week[iso].as<JsonArray>();
      int idx=0;
      for (JsonVariant v: times){
        if (idx>=MAX_MEALS_PER_DAY) break;
        String t=v.asString(); t.trim();
        if (isValidHHMM(t)) { dayTimes[wday][idx]=t; idx++; }
      }
      dayCount[wday]=idx;
    }
    ok=true;
  }

  resetAllFed();
  resetAllBuzzed();
  waitingEat=false; pendingEatWday=-1; pendingEatIndex=-1;

  savePlanJSONToNVS(server.arg("plain"));
  saveAllFlags();

  Serial.println("=== Plan cargado ===");
  Serial.printf("mode: %s  secondsOpen: %.2f s\n", mode.c_str(), secondsOpen);
  for (int d=0; d<7; d++){ Serial.printf("Dia %d: ", d); for(int i=0;i<dayCount[d];i++){ Serial.print(dayTimes[d][i]); if(i<dayCount[d]-1) Serial.print(", "); } Serial.println(); }

  enableCORS();
  if (ok) server.send(200,"text/plain","Plan cargado");
  else    server.send(400,"text/plain","Error al cargar plan");
}

// /clearPlan (POST)
// ===========================
void handleClearPlan() {
  for (int d=0; d<7; d++){
    dayCount[d]=0;
    for(int i=0;i<MAX_MEALS_PER_DAY;i++){
      fed[d][i]=false; eaten[d][i]=false; water[d][i]=false;
      buzzed[d][i]=false;
    }
    saveDayFlags(d);
  }
  lastFeedWday=-1; lastFeedIndex=-1; lastFeedTime="";
  waitingEat=false; pendingEatWday=-1; pendingEatIndex=-1;
  enableCORS(); server.send(200,"text/plain","Plan borrado");
}

// ===========================
// /getPlan (GET)
// ===========================
void handleGetPlan() {
  enableCORS();
  String json = loadPlanJSONFromNVS();
  if (json.length() == 0) {
    StaticJsonDocument<2500> doc;
    doc["ok"] = true;
    doc["mode"] = "per_day";
    doc["secondsOpen"] = secondsOpen;
    JsonArray week = doc.createNestedArray("weekplan");
    for (int iso=0; iso<7; ++iso) {
      int wday = (iso == 6) ? 0 : iso + 1;
      JsonArray arr = week.createNestedArray();
      for (int i=0; i<dayCount[wday]; ++i) { arr.add(dayTimes[wday][i]); }
    }
    String out; serializeJson(doc, out);
    server.send(200, "application/json", out);
    return;
  }
  server.send(200, "application/json", json);
}

// ===========================
// Setup
// ===========================
void setup() {
  Serial.begin(115200);

  servoMotor.attach(SERVO_PIN, 500, 2400);
  servoMotor.write(0);

  pinMode(PIR_PIN, INPUT);

  pinMode(PUMP_PIN, INPUT_PULLUP);
  delay(5);
  pinMode(PUMP_PIN, OUTPUT);
  pump_off();

  setupBuzzer();

  WiFi.begin(ssid, password);
  Serial.print("Conectando WiFi");
  while (WiFi.status()!=WL_CONNECTED){ delay(500); Serial.print("."); }
  Serial.println("\nWiFi OK: " + WiFi.localIP().toString());

  configTime(gmtOffset_sec, daylightOffset_sec, ntpServer);
  struct tm timeinfo;
  if (getLocalTime(&timeinfo)) { lastWday = timeinfo.tm_wday; Serial.printf("Hora inicial: %02d:%02d  wday=%d\n", timeinfo.tm_hour, timeinfo.tm_min, lastWday); }
  else                          { Serial.println("No se pudo obtener hora inicial (NTP)"); }

  // Restaurar plan desde NVS
  {
    String saved = loadPlanJSONFromNVS();
    if (saved.length() > 0) {
      StaticJsonDocument<2500> doc;
      DeserializationError err = deserializeJson(doc, saved);
      if (!err) {
        if (doc.containsKey("secondsOpen")) { secondsOpen = doc["secondsOpen"].as<float>(); }
        String mode = doc.containsKey("mode") ? String(doc["mode"].as<const char*>()) : "per_day";
        for (int d=0; d<7; d++){
          dayCount[d]=0;
          for(int i=0;i<MAX_MEALS_PER_DAY;i++){
            fed[d][i]=false; eaten[d][i]=false; water[d][i]=false;
            buzzed[d][i]=false;
          }
        }
        auto isoToWday = [](int iso)->int { return (iso==6)?0:(iso+1); };
        if (mode=="repeat_week" && doc.containsKey("dayplan")) {
          JsonArray arr = doc["dayplan"].as<JsonArray>();
          String tmp[MAX_MEALS_PER_DAY]; int cnt=0;
          for (JsonVariant v: arr){ if (cnt>=MAX_MEALS_PER_DAY) break; String t=v.as<String>(); t.trim(); if (isValidHHMM(t)) tmp[cnt++]=t; }
          for (int d=0; d<7; d++){ for (int i=0;i<cnt;i++){ dayTimes[d][i]=tmp[i]; } dayCount[d]=cnt; }
        } else if (mode=="per_day" && doc.containsKey("weekplan")) {
          JsonArray week = doc["weekplan"].as<JsonArray>();
          if (week.size()==7){
            for (int iso=0; iso<7; iso++){
              int wday=isoToWday(iso);
              JsonArray times = week[iso].as<JsonArray>();
              int idx=0;
              for (JsonVariant v: times){
                if (idx>=MAX_MEALS_PER_DAY) break;
                String t=v.as<String>(); t.trim();
                if (isValidHHMM(t)) { dayTimes[wday][idx]=t; idx++; }
              }
              dayCount[wday]=idx;
            }
          }
        }
        resetAllFed();
        resetAllBuzzed();
        waitingEat=false; pendingEatWday=-1; pendingEatIndex=-1;

        Serial.println("Plan restaurado desde NVS.");
      } else {
        Serial.println("No se pudo parsear plan NVS.");
      }
    } else {
      Serial.println("No hay plan NVS guardado.");
    }
  }

  // Cargar dueño de plan si existía
  if (prefs.begin("petcare", true)) {
    currentPlanUserId = prefs.getString("plan_user", "");
    prefs.end();
  }

  loadAllFlags();

  setupMicADC();
  server.on("/mic/start",  HTTP_GET,      [](){ enableCORS(); micEnabled=true; micStopAtMs = millis() + 15UL*60UL*1000UL; server.send(200,"application/json","{\"ok\":true}"); });
  server.on("/mic/stop",   HTTP_GET,      [](){ enableCORS(); micEnabled=false; micStopAtMs=0; server.send(200,"application/json","{\"ok\":true}"); });
  server.on("/mic/status", HTTP_GET,      handleStatus);
  server.on("/mic/start",  HTTP_OPTIONS,  handleOptions);
  server.on("/mic/stop",   HTTP_OPTIONS,  handleOptions);
  server.on("/mic/status", HTTP_OPTIONS,  handleOptions);

  server.on("/setPlan",   HTTP_POST,    handleSetPlan);
  server.on("/setPlan",   HTTP_OPTIONS, handleOptions);
  server.on("/status",    HTTP_GET,     handleStatus);
  server.on("/status",    HTTP_OPTIONS, handleOptions);
  server.on("/clearPlan", HTTP_POST,    handleClearPlan);
  server.on("/clearPlan", HTTP_OPTIONS, handleOptions);
  server.on("/servo",     HTTP_GET,     handleServoManual);
  server.on("/servo",     HTTP_OPTIONS, handleOptions);
  server.on("/getPlan",   HTTP_GET,     handleGetPlan);
  server.on("/getPlan",   HTTP_OPTIONS, handleOptions);

  // Buzzer
  server.on("/buzzer", HTTP_POST, []() {
    enableCORS();
    buzzerAvisoComida();
    server.send(200, "application/json", "{\"ok\":true}");
  });
  server.on("/buzzer", HTTP_OPTIONS, handleOptions);

  // NUEVO: Calibración y Agua
  server.on("/calibration/open", HTTP_GET,     handleCalibOpen);
  server.on("/calibration/open", HTTP_OPTIONS, handleOptions);
  server.on("/calibration",      HTTP_GET,     handleCalibGet);
  server.on("/calibration",      HTTP_POST,    handleCalibPost);
  server.on("/calibration",      HTTP_OPTIONS, handleOptions);

  server.on("/water",            HTTP_GET,     handleWaterGet);
  server.on("/water",            HTTP_POST,    handleWaterPost);
  server.on("/water",            HTTP_OPTIONS, handleOptions);

  server.begin();

  loadAllMicAlerts();
  struct tm _ti;
  if (getLocalTime(&_ti)) { micAlertsToday = micAlerts[_ti.tm_wday]; } else { micAlertsToday = 0; }
  xTaskCreatePinnedToCore(micTask, "micTask", 4096, nullptr, 1, &micTaskHandle, 0);
}

// ===========================
// Loop
// ===========================
unsigned long lastCheck = 0;
const unsigned long CHECK_EVERY_MS = 1000;

void loop() {
  server.handleClient();

  unsigned long nowMs = millis();
  if (nowMs - lastCheck >= CHECK_EVERY_MS) {
    lastCheck = nowMs;

    struct tm timeinfo;
    if (getLocalTime(&timeinfo)) {
      int wday = timeinfo.tm_wday;

      if (wday != lastWday) {
        resetFedForDay(wday);
        resetBuzzedForDay(wday);
        lastWday = wday;
        saveDayFlags(wday);
        Serial.printf("Cambio de dia: wday=%d, reset flags\n", wday);
      }

      char hmbuf[6]; strftime(hmbuf, sizeof(hmbuf), "%H:%M", &timeinfo);
      String nowHHMM = String(hmbuf);

      int count = dayCount[wday];
      for (int i=0; i<count; i++) {

        // Aviso 1 minuto antes
        if (!buzzed[wday][i]) {
          String pre = prevMinute(dayTimes[wday][i]);
          if (pre.length()==5 && nowHHMM == pre) {
            Serial.printf("Aviso buzzer para comida (wday=%d idx=%d) a las %s\n", wday, i, nowHHMM.c_str());
            buzzerAvisoComida();
            buzzed[wday][i] = true;
          }
        }

        if (!fed[wday][i] && nowHHMM == dayTimes[wday][i]) {
          fed[wday][i] = true;

          lastFeedWday  = wday;
          lastFeedIndex = i;
          lastFeedTime  = nowHHMM;

          Serial.printf("Alimentando (wday=%d) a las %s\n", wday, dayTimes[wday][i].c_str());

          // Servo
          servoMotor.write(90);
          delay((unsigned long)(secondsOpen * 1000));
          servoMotor.write(0);

          // Gap
          delay(PUMP_GAP_MS);

          // Bomba — calcular ms según ml y caudal por usuario (defaults si no hay)
          float flow = loadUserPumpFlow(currentPlanUserId);
          if (!isfinite(flow) || flow <= 0) flow = DEFAULT_PUMP_ML_PER_SEC;

          float mlPerMeal = loadUserWaterML(currentPlanUserId);
          if (!isfinite(mlPerMeal) || mlPerMeal < 0) mlPerMeal = DEFAULT_WATER_ML_PER_MEAL;

          unsigned long pumpMs = (unsigned long)((mlPerMeal / flow) * 1000.0f);

          pump_on();
          delay(pumpMs);
          pump_off();

          water[wday][i] = true;
          saveDayFlags(wday);

          waitingEat       = true;
          pendingEatWday   = wday;
          pendingEatIndex  = i;
          eatWindowStartMs = millis();
        }
      }
    }
  }

  if (waitingEat) {
    if (millis() - eatWindowStartMs >= EAT_WINDOW_MS) {
      waitingEat = false;
    } else {
      if (digitalRead(PIR_PIN) == HIGH) {
        if (pendingEatWday>=0 && pendingEatIndex>=0 &&
            pendingEatWday<=6 && pendingEatIndex<dayCount[pendingEatWday]) {
          eaten[pendingEatWday][pendingEatIndex] = true;
          Serial.printf("Comido detectado (wday=%d, idx=%d)\n", pendingEatWday, pendingEatIndex);
          saveDayFlags(pendingEatWday);
        }
        waitingEat = false;
      }
    }
  }

  static int _micDay = -1;
  struct tm ti2;
  if (getLocalTime(&ti2)) {
    if (_micDay != ti2.tm_wday) {
      if (_micDay >= 0 && _micDay <= 6) saveMicAlerts(_micDay);
      _micDay = ti2.tm_wday;
      micAlerts[_micDay] = 0;
      micAlertsToday = 0;
      micInAlert = false;
      saveMicAlerts(_micDay);
    }
  }
}
